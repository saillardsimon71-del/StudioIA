def seed_for(name: str) -> int:
    return int.from_bytes(hashlib.sha256(f"ep44v2-{name}".encode()).digest()[:8], "big")


def main() -> None:
    parser = argparse.ArgumentParser()
    parser.add_argument("--dry-run", action="store_true")
    parser.add_argument("--resume", action="store_true")
    parser.add_argument("--only", default="")
    args = parser.parse_args()
    engine = ModalH3Engine(settings.studio_modal_app, settings.studio_modal_function)
    timeline = []
    for name, image, duration, prompt in SHOTS:
        if args.only and not any(part in name for part in args.only.split(",")):
            continue
        target = OUT / name
        target.mkdir(parents=True, exist_ok=True)
        (target / "prompt.txt").write_text(prompt, encoding="utf-8")
        payload = build_request(
            prompt=prompt, duration_sec=duration, aspect_ratio="9:16",
            seed=seed_for(name), image_path=image, audio_mode="native",
            trace_id=f"ep44v2_{name}".replace("-", "_")[:64],
            video_profile="ltx",
        )
        safe = {key: value for key, value in payload.items() if key != "image_base64"}
        safe["image_path"] = str(image)
        (target / "request.json").write_text(
            json.dumps(safe, ensure_ascii=False, indent=2), encoding="utf-8",
        )
        print(f"[{name}] {len(prompt.split())} mots, {payload['frames']} frames", flush=True)
        if args.dry_run:
            continue
        if args.resume and (target / "video.mp4").is_file():
            print(f"[{name}] conserve", flush=True)
            timeline.append({"name": name, "duration": duration,
                             "path": str(target / "video.mp4")})
            continue
        call_id = engine.spawn(payload)
        print(f"[{name}] modal_call={call_id}", flush=True)
        result = None
        deadline = time.time() + 3600
        while result is None:
            result = engine.poll(call_id)
            if result is not None:
                break
            if engine.remote_terminal:
                raise RuntimeError(f"[{name}] termine sans resultat")
            if time.time() > deadline:
                raise TimeoutError(name)
            time.sleep(5)
        (target / "video.mp4").write_bytes(base64.b64decode(result["video_base64"]))
        (target / "metadata.json").write_text(
            json.dumps(result.get("metadata") or {}, indent=2), encoding="utf-8",
        )
        timeline.append({"name": name, "duration": duration,
                         "path": str(target / "video.mp4")})
        (OUT / "timeline.json").write_text(
            json.dumps(timeline, ensure_ascii=False, indent=2), encoding="utf-8",
        )
        print(f"[{name}] termine", flush=True)


if __name__ == "__main__":
    main()
